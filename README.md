## Основные команды
~~~C#
// Работа с консолью
Console.WriteLine("Текст");         // Выводит текст в консоль и переносит курсор на новую строку
Console.Write("Текст");             // Выводит текст в консоль без переноса строки
Console.ReadLine();                 // Считывает строку, введенную пользователем
Console.ReadKey();                  // Ожидает нажатия любой клавиши (часто используется, чтобы окно не закрылось)
Console.Clear();                    // Полностью очищает окно терминала
// создание переменных (типы данных)
int x = 10;                         // Целое число
double y = 3.14;                    // Число с плавающей точкой (аналог float в Python)
string s = "Hello";                 // Строка текста (обязательно в двойных кавычках)
bool b = true;                      // Логическое значение (true или false)
char c = 'A';                       // Один символ (обязательно в одинарных кавычках)
var z = 5;                          // Автоматическое определение типа компилятором	
// Конвертация типов
int.Parse("123");                   // Превращает строку в целое число
Convert.ToDouble(x);                // Превращает переменную в тип double
x.ToString();                       // Превращает любое значение или переменную в строку
// математика и логика
x++;                                // Инкремент (увеличение переменной на 1)
x--;                                // Декремент (уменьшение переменной на 1)
Math.Max(a, b);                     // Возвращает большее из двух чисел
Math.Abs(x);                        // Возвращает модуль (абсолютное значение) числа
// условия, управление, инструкции
if (условие) { }                    // Условный оператор (код внутри скобок выполняется, если условие истинно)
else { }                            // Выполняется, если условие в if было ложным
for (int i=0; i<10; i++) { }        // Цикл со счетчиком (старт; условие; шаг)
while (условие) { }                 // Цикл, который работает, пока условие истинно
break;                              // Команда для немедленного выхода из цикла
// списки
using System.Collections.Generic;
List<int> list = new List<int>();   // Создание нового пустого списка для целых чисел
list.Add(5);                        // Добавление элемента в список
list.RemoveAt(0);                   // Удаление элемента по его индексу
list.Count;                         // Получение количества элементов в списке (аналог len() в Python)
~~~

## ПРиколы для Bodhi Linux
~~~C#
// Специфические команды
tyls                                // Выводит список файлов в папке с миниатюрами картинок и видео (очень наглядно)
tyalpha on                          // Делает окно терминала прозрачным (удобно, когда код на фоне)
tyalpha off                         // Выключает прозрачность
typop [file]                        // Открывает файл (картинку или документ) во всплывающем окне прямо внутри терминала
tybg [image_path]                   // Устанавливает картинку на фон терминала
tycat [file]                        // Быстро отображает содержимое файла (текст или даже картинку) прямо в консоли
// РАбота с кодом C#
dotnet new console                  // Создать шаблон нового проекта "Hello World" в текущей папке
dotnet run                          // Скомпилировать и сразу запустить твою программу
dotnet build                        // Просто проверить код на ошибки и собрать его, не запуская
dotnet clean                        // Очистить проект от временных файлов сборки
dotnet new console                  // создает шаблон консольного приложения
-o <название>                       // создает папку для консольного приложения
~~~ 
### Управление окнами (терминала)
Ctrl + Shift + PgUp/PgDn            // Быстрое переключение между вкладками
Ctrl + Shift + Home                 // Разделить окно терминала по вертикали (чтобы в одном писать код, а в другом — запускать)
Ctrl + Shift + End                  // Разделить окно по горизонтали
Alt + Стрелки                       // Перемещение между разделенными частями окна
Ctrl + Shift + T                    // Создать новую вкладку (как в браузере)
